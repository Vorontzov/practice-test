  // 2. ordered dice sums
  function orderedTriples(sum,sides=6){let c=0;for(let a=1;a<=sides;a++)for(let b=1;b<=sides;b++)for(let d=1;d<=sides;d++)if(a+b+d===sum)c++;return c}
  function q2(){const sides=rnd(4,30),sum=rnd(4,3*sides-1),ans=orderedTriples(sum,sides);return makeShort('ordered-dice',{sides,sum},`<p>Three fair number cubes have sides labeled 1 through ${sides}.   function primeFactor(n){const out={};let d=2;while(n>1){while(n%d===0){out[d]=(out[d]||0)+1;n/=d}d++;if(d*d>n&&n>1){out[n]=(out[n]||0)+1;break}}return out}
  function divisorCount(n){return Object.values(primeFactor(n)).reduce((a,e)=>a*(e+1),1)}
  function divisorSum(n){return Object.entries(primeFactor(n)).reduce((prod,[p,e])=>prod*((Math.pow(+p,e+1)-1)/(+p-1)),1)}
  // 6 divisor count
